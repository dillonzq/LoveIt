{{- /* Icon markup built at build time, or "" when it cannot be (plugin/icon.html then falls back). */ -}} {{- /* Class: Font Awesome classes such as "fab fa-github" (legacy names work, see svgs/aliases.json) */ -}} {{- /* and theme classes kept on the SVG; size, animation and other fa-* modifiers are not supported. */ -}} {{- /* Simpleicons (with an optional Prefix) or Src: an SVG resource, inlined when local. */ -}} {{- /* Label: accessible name; without it the icon is decorative and hidden from assistive technology. */ -}} {{- /* Templates call plugin/icon.html, which caches the result. */ -}} {{- $resource := false -}} {{- $remote := "" -}} {{- $classes := slice "icon" -}} {{- with .Class -}} {{- $styles := dict "fa" "solid" "fas" "solid" "fa-solid" "solid" "far" "regular" "fa-regular" "regular" "fab" "brands" "fa-brands" "brands" -}} {{- $style := "" -}} {{- $names := slice -}} {{- range split (trim . " ") " " -}} {{- if index $styles . -}} {{- $style = index $styles . -}} {{- else if hasPrefix . "fa-" -}} {{- /* Font Awesome 7 icons are fixed width by default. */ -}} {{- if ne . "fa-fw" -}} {{- $names = $names | append (strings.TrimPrefix "fa-" .) -}} {{- end -}} {{- else if . -}} {{- $classes = $classes | append . -}} {{- end -}} {{- end -}} {{- if and $style (eq (len $names) 1) -}} {{- $name := index $names 0 -}} {{- $aliases := resources.Get "lib/fontawesome-free/svgs/aliases.json" | transform.Unmarshal -}} {{- $resource = printf "lib/fontawesome-free/svgs/%v/%v.svg" $style (index $aliases $name | default $name) | resources.Get -}} {{- end -}} {{- else -}} {{- $src := .Src -}} {{- with .Simpleicons -}} {{- $prefix := $.Prefix | default "lib/simple-icons/icons" | string | strings.TrimSuffix "/" -}} {{- $src = printf "%v/%v.svg" $prefix . -}} {{- /* The bundled copy is used even when a CDN prefix is configured; the CDN serves icons it lacks. */ -}} {{- if (urls.Parse $prefix).Host -}} {{- $resource = printf "lib/simple-icons/icons/%v.svg" . | resources.Get -}} {{- end -}} {{- end -}} {{- if not $resource | and $src -}} {{- if (urls.Parse (string $src)).Host -}} {{- $remote = $src -}} {{- else -}} {{- $resource = resources.Get (strings.TrimPrefix "/" $src) -}} {{- end -}} {{- end -}} {{- end -}} {{- $class := delimit (uniq $classes) " " | htmlEscape -}} {{- $label := .Label | default "" | htmlEscape -}} {{- $a11y := `aria-hidden="true"` -}} {{- with $label -}} {{- $a11y = printf `role="img" aria-label="%v"` . -}} {{- end -}} {{- $markup := "" -}} {{- with $resource -}} {{- $source := .Content -}} {{- $open := index (findRE `]*>` $source 1) 0 | default "" -}} {{- if findRE `\sviewBox\s*=` $open 1 -}} {{- /* Keep the root's presentation (viewBox, fill, stroke, style...) and classes; the theme sets */ -}} {{- /* size and accessibility, and fills with the text color unless the root sets a fill. */ -}} {{- with findRESubmatch `\sclass\s*=\s*(?:"([^"]*)"|'([^']*)')` $open 1 -}} {{- $class = add (index . 0 1) (index . 0 2) | htmlUnescape | htmlEscape | printf "%v %v" $class -}} {{- end -}} {{- $attrs := replaceRE `^$` "" $open | replaceRE `\s(class|width|height|role|aria-[\w-]+|xmlns)\s*=\s*("[^"]*"|'[^']*')` "" | strings.TrimSpace -}} {{- if not (findRE `(^|\s)fill\s*=|fill\s*:` $attrs 1) -}} {{- $attrs = printf `%v fill="currentColor"` $attrs -}} {{- end -}} {{- /* Keep the Font Awesome license comment (minified output drops it); drop titles, Label names the icon. */ -}} {{- $inner := replaceRE `(?s)^.*?]*>` "" $source 1 | replaceRE `(?s)\s*$` "" | replaceRE `(?s)` "" -}} {{- if $.Class -}} {{- /* Font Awesome paths repeat the root's fill. */ -}} {{- $inner = replaceRE `\s+fill="currentColor"` "" $inner -}} {{- end -}} {{- /* The size attributes keep the icon small where the theme's CSS does not apply (feed readers). */ -}} {{- $markup = printf `` $class $attrs $a11y (strings.TrimSpace $inner) -}} {{- else -}} {{- /* No viewBox to scale by: publish the file and show it as an image. */ -}} {{- $markup = printf `` $class .RelPermalink $label (cond (eq $label "") ` aria-hidden="true"` "") -}} {{- end -}} {{- else -}} {{- with $remote -}} {{- $markup = printf `` $class (htmlEscape .) $label (cond (eq $label "") ` aria-hidden="true"` "") -}} {{- end -}} {{- end -}} {{- return $markup -}}
