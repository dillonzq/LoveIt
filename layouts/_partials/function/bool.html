{{- /* To Bool */ -}} {{- /* true, false, "true", "false", "1" and "0" (in any case) become a boolean; */ -}} {{- /* other values, such as lineNos="inline", are returned as they are. */ -}} {{- $bool := . -}} {{- $value := . -}} {{- if eq (printf "%T" .) "string" -}} {{- $value = lower . -}} {{- end -}} {{- if eq $value true -}} {{- $bool = true -}} {{- else if eq $value false -}} {{- $bool = false -}} {{- else if eq $value "true" -}} {{- $bool = true -}} {{- else if eq $value "false" -}} {{- $bool = false -}} {{- else if eq $value "1" -}} {{- $bool = true -}} {{- else if eq $value "0" -}} {{- $bool = false -}} {{- end -}} {{- return $bool -}}
