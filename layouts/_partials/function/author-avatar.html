{{- /* The avatar of a page author. Input: dict "Page" (the page) "Params" (its parameters merged with params.page). authorAvatar in front matter, or else params.author.avatar when the page author (Params.author) is unset or is params.author.name, so that a page by another author does not show the site author's photo. Returns dict "Path" "Resources" for function/resource.html and plugin/img.html, or "" without an avatar. Only authorAvatar is looked up in the page resources; params.author.avatar is the same file on every page. A local path that is not a resource, such as a file under static/, goes through relURL, so a path without a leading slash keeps the path of baseURL. */ -}} {{- $page := .Page -}} {{- $author := .Params.author -}} {{- $avatar := "" -}} {{- $resources := "" -}} {{- with .Params.authoravatar -}} {{- $avatar = . -}} {{- $resources = $page.Resources -}} {{- else -}} {{- with $page.Site.Params.Author -}} {{- if not $author | or (eq (printf "%v" $author) (printf "%v" (.name | default ""))) -}} {{- $avatar = .avatar -}} {{- end -}} {{- end -}} {{- end -}} {{- $result := "" -}} {{- with $avatar -}} {{- $path := printf "%v" . -}} {{- $url := urls.Parse $path -}} {{- if not $url.Scheme | and (not $url.Host) -}} {{- if not (dict "Path" $path "Resources" $resources | partial "function/resource.html") -}} {{- $path = relURL $path -}} {{- end -}} {{- end -}} {{- $result = dict "Path" $path "Resources" $resources -}} {{- end -}} {{- return $result -}}
