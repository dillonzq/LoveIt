Returns "" without an avatar. */ -}} {{- $avatar := .Params.authoravatar -}} {{- if not $avatar -}} {{- with .Site.Params.Author -}} {{- if not $.Params.author | or (eq (printf "%v" $.Params.author) (printf "%v" (.name | default ""))) -}} {{- $avatar = .avatar -}} {{- end -}} {{- end -}} {{- end -}} {{- return ($avatar | default "") -}}
