{{- /* Renders a Markdown heading. With "Anchor", a non-empty heading with an ID starts with a link to itself. The link is named by the heading, so it needs no JavaScript or translation. */ -}} {{- $anchor := .Context.Anchor -}} {{- /* An empty heading would give its link no name. */ -}} {{- if and .Anchor $anchor (findRE `[^\s\p{Z}\x{0085}\x{000B}]` (htmlUnescape .Context.PlainText) 1) -}} {{- $label := printf "%s:%s" .Context.Page.Path $anchor | sha256 | printf "loveit-heading-%s" -}} {{ .Context.Text }} {{- else -}} {{- .Context.Text -}} {{- end -}}
