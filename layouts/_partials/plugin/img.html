{{- /* Render an image, optionally linked for lightGallery. Options: - Src: page resource name, path under assets/ or static/, or URL. - Resources: the page resources to look Src up in first. - Alt, Title, Caption, Class, Rel, Linked: as before. - Width, Height: override the intrinsic size. When only one is a number, the other follows the image's aspect ratio. - SrcSmall, SrcLarge: lightGallery thumbnail and full-size image. - Eager: an image in the first screen that may be the largest one (loading="eager" and fetchpriority="high"). - Lazy: false loads the image at once without raising its priority. Other images are lazy-loaded natively and decoded asynchronously. - Widths: widths of the WebP copies in srcset; an empty slice disables them. - Sizes: the sizes attribute that goes with srcset. Page resources and global resources (assets/) in JPEG, PNG or WebP get width, height and a srcset of WebP copies made with Hugo's image processing, followed by the original file when it is wider than the largest copy. GIF, BMP, TIFF and animated PNG or WebP only get their size; SVG, other formats, files under static/ and remote URLs are linked as they are. */ -}} {{- $src := .Src -}} {{- $width := .Width -}} {{- $height := .Height -}} {{- $widths := cond (isset . "Widths") .Widths (slice 480 960) -}} {{- $sizes := .Sizes | default "(max-width: 680px) 100vw, (max-width: 960px) 80vw, (max-width: 1280px) 52vw, (max-width: 1440px) 56vw, (max-width: 1688px) 64vw, 1080px" -}} {{- $candidates := slice -}} {{- $thumbnail := "" -}} {{- with dict "Path" $src "Resources" .Resources | partial "function/resource.html" -}} {{- $src = .RelPermalink -}} {{- $type := not when it resizes. */ -}} {{- $orientation := 1 -}} {{- if ge hugo.Version "0.155.3" -}} {{- if in (slice "jpeg" "png" "webp" "tiff") $type -}} {{- with .Meta -}} {{- $orientation = .Orientation | $orientation = int $orientation -}} {{- $imageWidth := .Width -}} {{- $imageHeight := .Height -}} {{- if ge $orientation 5 -}} {{- $imageWidth = .Height -}} {{- $imageHeight = .Width -}} {{- end -}} {{- /* Fill in a missing dimension from the aspect ratio when the other one is a number of pixels; 0 counts as missing. */ -}} {{- $w := string $width -}} {{- $h := string $height -}} {{- if eq $w "0" -}} {{- $w = "" -}} {{- $width = "" -}} {{- end -}} {{- if eq $h "0" -}} {{- $h = "" -}} {{- $height = "" -}} {{- end -}} {{- if and $w (not $h) (findRE `^\d+$` $w) -}} {{- $height = div (mul (float $w) $imageHeight) $imageWidth | math.Round | math.Max 1 | int -}} {{- else if and $h (not $w) (findRE `^\d+$` $h) -}} {{- $width = div (mul (float $h) $imageWidth) $imageHeight | math.Round | math.Max $animated = strings.Contains (index (split (index (split .Content "VP8 ") 0) "VP8L") 0) "ANIM" -}} {{- end -}} {{- /* WebP encoding needs the extended edition before Hugo 0.153.0. */ -}} {{- if in (slice "jpeg" "png" "webp") $type | and (not $animated) (or hugo.IsExtended (ge hugo.Version "0.153.0")) -}} {{- $image := . -}} {{- if gt $orientation 1 -}} {{- $image = $image.Filter images.AutoOrient -}} {{- end -}} {{- /* Copies narrower than the image, plus one at its own width when a width reaches it. */ -}} {{- $copies := slice -}} {{- range $widths -}} {{- if lt . $imageWidth -}} {{- $copies = $copies | append . -}} {{- end -}} {{- end -}} {{- $covered := lt (len $copies) (len $widths) -}} {{- if $covered -}} {{- $copies = $copies | append $imageWidth -}} {{- end -}} {{- /* An image Hugo cannot decode keeps only its size instead of failing the build. */ -}} {{- $failed := false -}} {{- range sort ($copies | uniq) -}} {{- with try ($image.Resize (printf {{- end -}} {{- end -}} {{- if $failed -}} {{- $candidates = slice -}} {{- else if $candidates -}} {{- $thumbnail = (index $candidates 0).URL -}} {{- if not $covered -}} {{- $candidates = $candidates
